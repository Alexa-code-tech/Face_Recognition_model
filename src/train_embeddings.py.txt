"""
Train Face Embeddings Script

Extracts and saves face embeddings from known faces
"""

import os
import argparse
import logging
from typing import Dict
import numpy as np

from models.recognition_pipeline import RecognitionPipeline
from src.utils import (
    setup_logging, get_face_directories, 
    save_embeddings, time_function
)

logger = logging.getLogger(__name__)


@time_function
def train_embeddings(known_faces_dir: str,
                    output_path: str,
                    detector_name: str = "mtcnn",
                    embedder_name: str = "facenet",
                    distance_metric: str = "euclidean",
                    tolerance: float = 0.6,
                    device: str = "cuda") -> int:
    """
    Обучить и сохранить эмбеддинги для известных лиц
    
    Args:
        known_faces_dir: Директория с известными лицами
        output_path: Путь для сохранения эмбеддингов
        detector_name: Название детектора
        embedder_name: Название эмбедера
        distance_metric: Метрика расстояния
        tolerance: Порог подобия
        device: cuda или cpu
        
    Returns:
        Количество успешно обработанных лиц
    """
    
    # Инициализировать pipeline
    pipeline = RecognitionPipeline(
        detector_name=detector_name,
        embedder_name=embedder_name,
        distance_metric=distance_metric,
        tolerance=tolerance,
        device=device
    )
    
    # Получить структуру директорий
    face_dirs = get_face_directories(known_faces_dir)
    
    if not face_dirs:
        logger.error(f"No faces found in {known_faces_dir}")
        return 0
    
    logger.info(f"Found {len(face_dirs)} people")
    
    # Зарегистрировать каждого человека
    total_registered = 0
    for name, image_paths in face_dirs.items():
        registered = pipeline.register_faces(name, image_paths)
        total_registered += registered
        logger.info(f"{name}: {registered}/{len(image_paths)} faces registered")
    
    # Сохранить матчер с эмбеддингами
    embeddings_dict = pipeline.matcher.known_embeddings
    save_embeddings(embeddings_dict, output_path)
    
    logger.info(f"Total: {total_registered} faces registered and saved")
    
    return total_registered


def main():
    parser = argparse.ArgumentParser(description="Train face embeddings")
    parser.add_argument("--known_faces_dir", type=str, default="data/known_faces",
                       help="Path to directory with known faces")
    parser.add_argument("--output", type=str, default="data/embeddings/face_embeddings.pkl",
                       help="Output path for embeddings")
    parser.add_argument("--detector", type=str, default="mtcnn",
                       choices=["mtcnn", "retinaface", "dlib"],
                       help="Face detector to use")
    parser.add_argument("--embedder", type=str, default="facenet",
                       choices=["facenet", "vggface2", "openface"],
                       help="Face embedder to use")
    parser.add_argument("--distance_metric", type=str, default="euclidean",
                       choices=["euclidean", "cosine"],
                       help="Distance metric for matching")
    parser.add_argument("--tolerance", type=float, default=0.6,
                       help="Tolerance threshold for face matching")
    parser.add_argument("--device", type=str, default="cuda",
                       choices=["cuda", "cpu"],
                       help="Device to use (cuda or cpu)")
    parser.add_argument("--log_level", type=str, default="INFO",
                       help="Logging level")
    
    args = parser.parse_args()
    
    # Настроить логирование
    setup_logging(level=args.log_level)
    
    # Обучить эмбеддинги
    train_embeddings(
        known_faces_dir=args.known_faces_dir,
        output_path=args.output,
        detector_name=args.detector,
        embedder_name=args.embedder,
        distance_metric=args.distance_metric,
        tolerance=args.tolerance,
        device=args.device
    )


if __name__ == "__main__":
    main()
